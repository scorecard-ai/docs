You'll write one script that runs a small "tone rewriter" on three testcases, scores each output with an AI metric, and prints a link to the results. Swap in your own agent when you're ready.

<Steps>
  <Step title="Install the SDK and set your keys">
    <CodeGroup>
    ```bash Python
    pip install scorecard-ai
    ```

    ```bash JavaScript
    npm install scorecard-ai
    ```
    </CodeGroup>

    <ScorecardCredentials />
  </Step>

  <Step title="Write the evaluation script">
    The script has four parts: the system you're testing, the testcases, a metric, and a call to `run_and_evaluate`. Save it as `eval.py` or `eval.mjs`:

    <CodeGroup>
    ```python Python
    import os

    from scorecard_ai import Scorecard
    from scorecard_ai.lib import run_and_evaluate

    scorecard = Scorecard()  # reads SCORECARD_API_KEY
    PROJECT_ID = os.environ["SCORECARD_PROJECT_ID"]


    # 1. The system under test. It receives a testcase's inputs and returns a dict of outputs.
    #    This placeholder echoes the message back, so expect low scores. Replace it with your agent.
    def run_system(inputs: dict, _system_version) -> dict:
        return {"rewritten": inputs["original"]}


    # 2. Testcases: inputs for your system, plus the expected (ideal) output for the judge.
    testcases = [
        {
            "inputs": {"original": "We need your feedback on the new designs ASAP.", "tone": "polite"},
            "expected": {"idealRewritten": "Could you share your feedback on the new designs when you have a moment?"},
        },
        {
            "inputs": {"original": "I'll be late because my cat is asleep on my keyboard.", "tone": "funny"},
            "expected": {"idealRewritten": "Running late: my cat has claimed my keyboard and I'm awaiting her permission to leave."},
        },
        {
            "inputs": {"original": "pls send the report", "tone": "formal"},
            "expected": {"idealRewritten": "Could you please send me the report?"},
        },
    ]

    # 3. An AI metric. Reuse it on later runs instead of creating a duplicate.
    METRIC_NAME = "Tone accuracy"
    existing = {metric.name: metric for metric in scorecard.metrics.list(PROJECT_ID)}
    metric = existing.get(METRIC_NAME) or scorecard.metrics.create(
        PROJECT_ID,
        name=METRIC_NAME,
        eval_type="ai",
        output_type="int",
        prompt_template=(
            "Rate from 1 to 5 how well the response matches the requested tone: {{ inputs.tone }}.\n\n"
            "Original message: {{ inputs.original }}\n"
            "Response: {{ outputs.rewritten }}\n"
            "Ideal response: {{ expected.idealRewritten }}\n\n"
            "{{ gradingInstructionsAndExamples }}"
        ),
    )

    # 4. Run the system on every testcase, record the outputs, and score them.
    run = run_and_evaluate(
        client=scorecard,
        project_id=PROJECT_ID,
        testcases=testcases,
        metric_ids=[metric.id],
        system=run_system,
    )
    print(f"Results: {run['url']}")
    ```

    ```javascript JavaScript
    import Scorecard, { runAndEvaluate } from "scorecard-ai";

    const scorecard = new Scorecard(); // reads SCORECARD_API_KEY
    const PROJECT_ID = process.env.SCORECARD_PROJECT_ID;

    // 1. The system under test. It receives a testcase's inputs and returns an object of outputs.
    //    This placeholder echoes the message back, so expect low scores. Replace it with your agent.
    async function runSystem(inputs) {
      return { rewritten: inputs.original };
    }

    // 2. Testcases: inputs for your system, plus the expected (ideal) output for the judge.
    const testcases = [
      {
        inputs: { original: "We need your feedback on the new designs ASAP.", tone: "polite" },
        expected: { idealRewritten: "Could you share your feedback on the new designs when you have a moment?" },
      },
      {
        inputs: { original: "I'll be late because my cat is asleep on my keyboard.", tone: "funny" },
        expected: { idealRewritten: "Running late: my cat has claimed my keyboard and I'm awaiting her permission to leave." },
      },
      {
        inputs: { original: "pls send the report", tone: "formal" },
        expected: { idealRewritten: "Could you please send me the report?" },
      },
    ];

    // 3. An AI metric. Reuse it on later runs instead of creating a duplicate.
    const METRIC_NAME = "Tone accuracy";
    let metric;
    for await (const m of scorecard.metrics.list(PROJECT_ID)) {
      if (m.name === METRIC_NAME) metric = m;
    }
    metric ??= await scorecard.metrics.create(PROJECT_ID, {
      name: METRIC_NAME,
      evalType: "ai",
      outputType: "int",
      promptTemplate:
        "Rate from 1 to 5 how well the response matches the requested tone: {{ inputs.tone }}.\n\n" +
        "Original message: {{ inputs.original }}\n" +
        "Response: {{ outputs.rewritten }}\n" +
        "Ideal response: {{ expected.idealRewritten }}\n\n" +
        "{{ gradingInstructionsAndExamples }}",
    });

    // 4. Run the system on every testcase, record the outputs, and score them.
    const run = await runAndEvaluate(scorecard, {
      projectId: PROJECT_ID,
      testcases,
      metricIds: [metric.id],
      system: runSystem,
    });
    console.log(`Results: ${run.url}`);
    ```
    </CodeGroup>

    The metric's prompt template uses Jinja. `{{ inputs.* }}` and `{{ expected.* }}` come from the testcase, `{{ outputs.* }}` from your system, and `{{ gradingInstructionsAndExamples }}` adds Scorecard's scoring format. See [Metrics](/features/metrics) for every variable.
  </Step>

  <Step title="Run it">
    <CodeGroup>
    ```bash Python
    python eval.py
    ```

    ```bash JavaScript
    node eval.mjs
    ```
    </CodeGroup>

    The script prints a link to the run. Scores appear within a minute as the judge finishes.
  </Step>

  <Step title="Plug in your agent">
    Replace `run_system` with a call to your agent and run the script again. For example, with OpenAI, install the client and set your key:

    <CodeGroup>
    ```bash Python
    pip install openai
    export OPENAI_API_KEY="sk-..."
    ```

    ```bash JavaScript
    npm install openai
    export OPENAI_API_KEY="sk-..."
    ```
    </CodeGroup>

    Then replace the placeholder:

    <CodeGroup>
    ```python Python
    from openai import OpenAI

    client = OpenAI()


    def run_system(inputs: dict, _system_version) -> dict:
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": f"Rewrite the user's message in a {inputs['tone']} tone."},
                {"role": "user", "content": inputs["original"]},
            ],
        )
        return {"rewritten": response.choices[0].message.content}
    ```

    ```javascript JavaScript
    import OpenAI from "openai";

    const client = new OpenAI();

    async function runSystem(inputs) {
      const response = await client.chat.completions.create({
        model: "gpt-4o-mini",
        messages: [
          { role: "system", content: `Rewrite the user's message in a ${inputs.tone} tone.` },
          { role: "user", content: inputs.original },
        ],
      });
      return { rewritten: response.choices[0].message.content };
    }
    ```
    </CodeGroup>

    Both runs are saved. To put them side by side, see [Compare runs](/features/a-b-comparison#compare-runs).
  </Step>
</Steps>

<Note>
  This metric uses `gpt-4o` as its judge. Without an OpenAI key under **Settings → AI Providers**, Scorecard uses its default judge model instead. See [Model providers](/features/custom-provider).
</Note>
